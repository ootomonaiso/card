#   Python 3.11 標準の tkinter / sqlite3 のみで動作（デバイスはモック）。
#
# Raspberry Pi 4 / Bookworm 実機でのみ以下が必要:
#   sudo apt install python3-tk        # Tkinter（未導入の場合）
#   pip install nfcpy                  # RC-S380 カード読取（FR-01）
#   pip install gpiozero               # GPIO 解錠制御（FR-05）
#
# nfcpy / gpiozero は実機デバイスを使うときだけ import されるため、
# 開発機では未インストールでも起動する。
nfcpy>=1.0.4 ; platform_machine == "aarch64" or platform_machine == "armv7l"
gpiozero>=2.0 ; platform_machine == "aarch64" or platform_machine == "armv7l"
